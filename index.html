require("dotenv").config() const express = require("express") const Database = require("better-sqlite3") const bcrypt = require("bcrypt") const jwt = require("jsonwebtoken") const https = require("https") const path = require("path") const { v4: uuidv4 } = require("uuid") const cors = require("cors") const app = express() app.use(cors()) app.use(express.json()) app.use(express.static("public")) const db = new Database("database.db") // ================= TABLES ================= db.prepare(` CREATE TABLE IF NOT EXISTS users ( id INTEGER PRIMARY KEY AUTOINCREMENT, email TEXT UNIQUE, password TEXT ) `).run() db.prepare(` CREATE TABLE IF NOT EXISTS downloads ( id TEXT PRIMARY KEY, url TEXT, expires_at INTEGER, downloads_count INTEGER DEFAULT 0 ) `).run() // ================= AUTH ================= function auth(req,res,next){ const token = req.headers.authorization?.split(" ")[1] if(!token) return res.status(401).json({error:"No token"}) try{ req.user = jwt.verify(token,process.env.JWT_SECRET) next() }catch{ res.status(403).json({error:"Invalid token"}) } } // ================= REGISTER ================= app.post("/register",(req,res)=>{ const {email,password} = req.body const hash = bcrypt.hashSync(password,10) try{ db.prepare("INSERT INTO users (email,password) VALUES (?,?)") .run(email,hash) res.json({message:"Conta criada"}) }catch{ res.status(400).json({error:"User exists"}) } }) // ================= LOGIN ================= app.post("/login",(req,res)=>{ const {email,password} = req.body const user = db.prepare("SELECT * FROM users WHERE email=?").get(email) if(!user) return res.status(404).json({error:"User not found"}) if(!bcrypt.compareSync(password,user.password)) return res.status(401).json({error:"Wrong password"}) const token = jwt.sign(user,process.env.JWT_SECRET) res.json({token}) }) // ================= FILES ================= const files = { "editor-pro": "https://lukintosh.com/downloads/editor.zip", "engine": "https://lukintosh.com/downloads/engine.zip", "cloud": "https://lukintosh.com/downloads/cloud.zip" } // ================= REQUEST ================= app.post("/request-trial",auth,(req,res)=>{ const fileUrl = files[req.body.product] if(!fileUrl) return res.status(400).json({error:"Invalid product"}) const id = uuidv4() db.prepare(` INSERT INTO downloads (id,url,expires_at) VALUES (?,?,?) `).run(id,fileUrl,Date.now()+1800000) res.json({ link: process.env.BASE_URL + "/download/" + id }) }) // ================= DOWNLOAD PAGE ================= app.get("/download/:id",(req,res)=>{ res.sendFile(path.join(__dirname,"public/download.html")) }) // ================= DOWNLOAD ================= app.get("/download-file/:id",(req,res)=>{ const row = db.prepare("SELECT * FROM downloads WHERE id=?") .get(req.params.id) if(!row) return res.send("Invalid") if(Date.now() > row.expires_at) return res.send("Expired") https.get(row.url,(r)=>{ r.pipe(res) }) }) // ================= HOME ================= app.get("/",(req,res)=>{ res.send("API ONLINE 🚀") }) app.listen(process.env.PORT || 3000)
